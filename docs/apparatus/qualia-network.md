function step(){
  if(alpha<.002)return; alpha*=.986;
  const ids=[...A];
  for(let i=0;i<ids.length;i++){const a=P[ids[i]];
    for(let j=i+1;j<ids.length;j++){const b=P[ids[j]];
      let dx=b.x-a.x,dy=b.y-a.y,d2=dx*dx+dy*dy;if(d2<1)d2=1;
      if(d2>230000)continue;
      const f=2600/d2,d=Math.sqrt(d2),fx=dx/d*f,fy=dy/d*f;
      a.vx-=fx;a.vy-=fy;b.vx+=fx;b.vy+=fy;}}
  EA.forEach(e=>{const a=P[e.source],b=P[e.target];
    const dx=b.x-a.x,dy=b.y-a.y,d=Math.hypot(dx,dy)||1;
    const rest=e.relation==='CARRIES'?86:118;
    const f=(d-rest)*0.028,fx=dx/d*f,fy=dy/d*f;
    a.vx+=fx;a.vy+=fy;b.vx-=fx;b.vy-=fy;});
  ids.forEach(id=>{const p=P[id];
    p.vx-=p.x*0.0032;p.vy-=p.y*0.0032;
    p.vx*=.86;p.vy*=.86;
    if(id!==dragId){p.x+=p.vx*alpha*2.1;p.y+=p.vy*alpha*2.1}});
}
